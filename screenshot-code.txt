        // ===== 生成详细目录 =====
        function generateDetailedTOC() {
            const tocList = document.getElementById('toc-list');
            if (!tocList) return;

            const contentArea = document.querySelector('.exercise-content-main');
            if (!contentArea) return;

            // 查找所有带 id 的标题元素
            const headings = contentArea.querySelectorAll('h3, h4, h5, h6');
            
            let tocHTML = '';
            headings.forEach((heading, index) => {
                const level = heading.tagName.toLowerCase();
                const text = heading.textContent;
                
                // 确保每个标题都有 id
                let id = heading.id;
                if (!id) {
                    id = `section-${level}-${index}`;
                    heading.id = id;
                }

                tocHTML += `<li class="${level}"><a href="#${id}" onclick="scrollToSection(event, '${id}')">${text}</a></li>`;
            });

            tocList.innerHTML = tocHTML;

            // 添加滚动监听以高亮当前章节
            setupScrollSpy();
        }

        // ===== 滚动监听（Scroll Spy） =====
        function setupScrollSpy() {
            const sections = document.querySelectorAll('[id^="section-"]');
            const tocLinks = document.querySelectorAll('.toc-list a');

            window.addEventListener('scroll', () => {
                let current = '';
                sections.forEach(section => {
                    const sectionTop = section.offsetTop;
                    const sectionHeight = section.clientHeight;
                    if (window.pageYOffset >= sectionTop - 150) {
                        current = section.getAttribute('id');
                    }
                });

                tocLinks.forEach(link => {
                    link.classList.remove('active');
                    if (link.getAttribute('href') === `#${current}`) {
                        link.classList.add('active');
                    }
                });
            });
        }

        // ===== 平滑滚动到指定章节 =====
        function scrollToSection(event, sectionId) {
            event.preventDefault();
            const section = document.getElementById(sectionId);
            if (section) {
                const offsetTop = section.offsetTop - 100; // 减去导航栏高度
                window.scrollTo({
                    top: offsetTop,
                    behavior: 'smooth'
                });
                
                // 更新 URL hash（不触发页面跳转）
                history.pushState(null, null, `#${sectionId}`);
            }
        }

        // ===== 路由管理 =====
        function handleRoute() {
            const hash = window.location.hash || '#home';
            
            // 隐藏所有页面
            document.querySelectorAll('.page-section').forEach(section => {
                section.classList.remove('active');
            });

            // 更新导航栏激活状态
            document.querySelectorAll('.nav-links a').forEach(link => {
                link.classList.remove('active');
                if (link.getAttribute('href') === hash || 
                    (hash.startsWith('#member-') && link.getAttribute('href') === '#members')) {
                    link.classList.add('active');
                }
            });

            // 处理成员详情页
            if (hash.startsWith('#member-')) {
                const memberId = parseInt(hash.split('-')[1]);
                if (!isNaN(memberId) && memberId >= 0 && memberId < members.length) {
                    showMemberDetailFromHash(memberId);
                    return;
                }
            }

            // 处理练习详情页
            if (hash.startsWith('#exercise-')) {
                const exerciseId = parseInt(hash.split('-')[1]);
                if (!isNaN(exerciseId) && exerciseId >= 0 && exerciseId < exercises.length) {
                    showExerciseDetailFromHash(exerciseId);
                    return;
                }
            }

            // 显示对应页面
            const targetSection = document.querySelector(hash);
            if (targetSection) {
                targetSection.classList.add('active');
            } else {
                // 无效路由重定向到首页
                window.location.hash = '#home';
            }

            // 滚动到顶部
            window.scrollTo(0, 0);
        }

        // ===== 滚动处理（导航栏隐藏 + 置顶按钮） =====
        let lastScrollY = window.scrollY;
        let ticking = false;

        function initScrollHandlers() {
            const navbar = document.querySelector('.navbar');
            const backToTopBtn = document.getElementById('backToTop');

            window.addEventListener('scroll', () => {
                if (!ticking) {
                    window.requestAnimationFrame(() => {
                        handleScroll(navbar, backToTopBtn);
                        ticking = false;
                    });
                    ticking = true;
                }
            });
        }

        function handleScroll(navbar, backToTopBtn) {
            const currentScrollY = window.scrollY;
            
            // 导航栏隐藏/显示逻辑
            if (currentScrollY > lastScrollY && currentScrollY > 100) {
                // 向下滚动且超过 100px，隐藏导航栏
                navbar.classList.add('hidden');
            } else {
                // 向上滚动或在顶部，显示导航栏
                navbar.classList.remove('hidden');
            }
            
            // 置顶按钮显示/隐藏逻辑
            if (currentScrollY > 300) {
                // 滚动超过 300px，显示置顶按钮
                backToTopBtn.classList.add('visible');
            } else {
                // 在顶部附近，隐藏置顶按钮
                backToTopBtn.classList.remove('visible');
            }
            
            lastScrollY = currentScrollY;
        }

        // ===== 平滑滚动到顶部 =====
        function scrollToTop() {
            window.scrollTo({
                top: 0,
                behavior: 'smooth'
            });
        }
