// JavaScript - Generate TOC dynamically
function generateDetailedTOC(exerciseId) {
    const tocList = document.getElementById(`toc-list-${exerciseId}`);
    if (!tocList) return;
    
    const homeworkSection = exerciseDetailContent.querySelector('.homework-content-section');
    if (!homeworkSection) return;
    
    const headings = homeworkSection.querySelectorAll('h4, h5');
    let tocHTML = `
        <li><a href="#section-exercise-info">Exercise ${exerciseId + 1}</a></li>
        <li><a href="#section-exercise-content">Exercise Content</a></li>
        <li><a href="#section-homework" class="active">Assignment Content</a></li>
    `;
    
    headings.forEach((heading, index) => {
        const level = heading.tagName === 'H4' ? '' : 'toc-sub';
        const text = heading.textContent.trim();
        const id = heading.id || `section-heading-${index}`;
        
        if (!heading.id) {
            heading.id = id;
        }
        
        tocHTML += `<li class="${level}"><a href="#${id}">${text}</a></li>`;
    });
    
    tocList.innerHTML = tocHTML;
    setupScrollSpy();
}
